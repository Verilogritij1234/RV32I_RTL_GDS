// dsp_chip_top.v
// Top-Level Padframe Wrapper for the DSP Engine

module dsp_chip_top (
    // External Physical Pins (These are what you solder to on the PCB)
    inout  PAD_CLK,
    inout  PAD_RESET,
    inout  PAD_START,
    inout  [15:0] PAD_DATA_IN_A,
    inout  [15:0] PAD_DATA_IN_B,
    
    inout  PAD_BUSY,
    inout  PAD_VALID,
    inout  PAD_SERIAL_OUT,
    
    // Core Power and Ground Pads
    inout  VDD_PAD, // 1.8V External Supply
    inout  VSS_PAD  // 0V External Ground
);

    // Internal wires connecting the Pad Ring to the Core
    wire core_clk;
    wire core_reset;
    wire core_start;
    wire [15:0] core_data_in_a;
    wire [15:0] core_data_in_b;
    
    wire core_busy;
    wire core_valid;
    wire core_serial_out;

    // =========================================================================
    // 1. INPUT PADS (Real World -> Core)
    // Using Sky130 standard GPIO pads configured as inputs
    // =========================================================================
    
    sky130_fd_io__top_gpiov2 PAD_IN_CLK (
        .PAD(PAD_CLK), .Y(core_clk), .INOUT_EN(1'b1), .IN(1'b0), .OUT_EN(1'b0)
    );
    
    sky130_fd_io__top_gpiov2 PAD_IN_RESET (
        .PAD(PAD_RESET), .Y(core_reset), .INOUT_EN(1'b1), .IN(1'b0), .OUT_EN(1'b0)
    );
    
    sky130_fd_io__top_gpiov2 PAD_IN_START (
        .PAD(PAD_START), .Y(core_start), .INOUT_EN(1'b1), .IN(1'b0), .OUT_EN(1'b0)
    );

    // Instantiate 16 Input Pads for DATA_IN_A
    genvar i;
    generate
        for (i = 0; i < 16; i = i + 1) begin : gen_pad_a
            sky130_fd_io__top_gpiov2 PAD_IN_A (
                .PAD(PAD_DATA_IN_A[i]), .Y(core_data_in_a[i]), .INOUT_EN(1'b1), .IN(1'b0), .OUT_EN(1'b0)
            );
        end
    endgenerate

    // Instantiate 16 Input Pads for DATA_IN_B
    generate
        for (i = 0; i < 16; i = i + 1) begin : gen_pad_b
            sky130_fd_io__top_gpiov2 PAD_IN_B (
                .PAD(PAD_DATA_IN_B[i]), .Y(core_data_in_b[i]), .INOUT_EN(1'b1), .IN(1'b0), .OUT_EN(1'b0)
            );
        end
    endgenerate

    // =========================================================================
    // 2. OUTPUT PADS (Core -> Real World)
    // Using Sky130 standard GPIO pads configured as outputs
    // =========================================================================

    sky130_fd_io__top_gpiov2 PAD_OUT_BUSY (
        .PAD(PAD_BUSY), .IN(core_busy), .INOUT_EN(1'b0), .OUT_EN(1'b1)
    );

    sky130_fd_io__top_gpiov2 PAD_OUT_VALID (
        .PAD(PAD_VALID), .IN(core_valid), .INOUT_EN(1'b0), .OUT_EN(1'b1)
    );

    sky130_fd_io__top_gpiov2 PAD_OUT_SERIAL (
        .PAD(PAD_SERIAL_OUT), .IN(core_serial_out), .INOUT_EN(1'b0), .OUT_EN(1'b1)
    );

    // =========================================================================
    // 3. POWER PADS
    // =========================================================================
    
    sky130_fd_io__top_power_lvc_wpad PAD_POWER (
        .VDDIO(VDD_PAD)
    );
    
    sky130_fd_io__top_ground_lvc_wpad PAD_GROUND (
        .VSSIO(VSS_PAD)
    );

    // =========================================================================
    // 4. INSTANTIATE THE CORE MACRO
    // =========================================================================
    
    dsp_engine core_inst (
        .clk(core_clk),
        .reset(core_reset),
        .start(core_start),
        .data_in_a(core_data_in_a),
        .data_in_b(core_data_in_b),
        .busy(core_busy),
        .valid(core_valid),
        .serial_out(core_serial_out)
    );

endmodule
