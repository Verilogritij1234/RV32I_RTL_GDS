proc place_my_pad {inst_name x_um y_um orient} {
    set block [ord::get_db_block]
    set dbu [$block getDefUnits]
    # Handle Yosys escaped names cleanly
    set inst [$block findInst $inst_name]
    if {$inst == "NULL"} {
        set inst [$block findInst "\\$inst_name "]
    }
    if {$inst != "NULL"} {
        $inst setOrient $orient
        $inst setOrigin [expr {int($x_um * $dbu)}] [expr {int($y_um * $dbu)}]
        $inst setPlacementStatus "FIRM"
    } else {
        puts "WARNING: Pad $inst_name not found in netlist!"
    }
}
